cmake_minimum_required(VERSION 4.0)

project(havOPL VERSION 0.1.0 LANGUAGES CXX)

add_library(havOPL STATIC havOPL/havOPL.cpp havOPL/havOPL.hpp)
add_library(havOPL::havOPL ALIAS havOPL)
target_include_directories(havOPL PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/havOPL)
target_compile_features(havOPL PUBLIC cxx_std_20)
set_target_properties(havOPL PROPERTIES CXX_EXTENSIONS OFF)

function(havopl_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()
havopl_warnings(havOPL)

# The tests build when havOPL is the top-level project (not when another project adds it)
option(HAVOPL_BUILD_TESTS "Build the havOPL tests" ${PROJECT_IS_TOP_LEVEL})
if(HAVOPL_BUILD_TESTS)
    enable_testing()
    foreach(test IN ITEMS havOPLTests havOPLEnvelopeTests havOPLModulationTests)
        add_executable(${test} tests/${test}.cpp)
        target_link_libraries(${test} PRIVATE havOPL::havOPL)
        set_target_properties(${test} PROPERTIES CXX_EXTENSIONS OFF)
        havopl_warnings(${test})
        add_test(NAME ${test} COMMAND ${test})
    endforeach()
endif()
